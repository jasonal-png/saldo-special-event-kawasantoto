const prizes = [ "50RB","100RB","150RB","200RB","250RB", "300RB","350RB","400RB","450RB","DANA 50RB" ]; const colors = [ "#ff79c9","#8d76ff","#48c9ee","#ffd36b", "#72dfb0","#ff9b65","#fa668b","#a77bff", "#50d5d0","#f6a5e5" ]; const canvas = document.getElementById("wheel"); const ctx = canvas.getContext("2d"); const center = 350; const radius = 330; const arc = Math.PI * 2 / prizes.length; let rotation = 0; let spinning = false; const STORAGE_KEY = "kawasantoto_spin_used"; function drawWheel() { ctx.clearRect(0,0,700,700); prizes.forEach((prize,i)=>{ const start = -Math.PI/2 + i*arc + rotation; const end = start + arc; ctx.beginPath(); ctx.moveTo(center,center); ctx.arc(center,center,radius,start,end); ctx.closePath(); ctx.fillStyle = colors[i]; ctx.fill(); ctx.strokeStyle = "#fff"; ctx.lineWidth = 5; ctx.stroke(); ctx.save(); ctx.translate(center,center); ctx.rotate(start+arc/2); ctx.textAlign = "right"; ctx.fillStyle = "#49204f"; ctx.font = "bold 27px Poppins, sans-serif"; ctx.fillText(prize,radius-28,9); ctx.restore(); }); } function showPreviousResult() { const saved = localStorage.getItem(STORAGE_KEY); if(saved !== null) { const data = JSON.parse(saved); amount.textContent = data.prize; result.style.display = "block"; button.disabled = true; button.textContent = "🔒 SPIN SUDAH DIGUNAKAN"; } } drawWheel(); showPreviousResult(); button.addEventListener("click",()=>{ if(spinning || localStorage.getItem(STORAGE_KEY)) return; spinning = true; button.disabled = true; button.textContent = "🎡 SEDANG BERPUTAR..."; result.style.display = "none"; const winner = Math.floor(Math.random()*prizes.length); const startRotation = rotation; const targetRotation = ((-(winner+0.5)*arc)%(Math.PI*2)+(Math.PI*2))%(Math.PI*2); const current = rotation%(Math.PI*2); const delta = (targetRotation-current+Math.PI*2)%(Math.PI*2); const endRotation = rotation + 10*Math.PI*2 + delta; const startTime = performance.now(); const duration = 5000; function animate(now) { const t = Math.min((now-startTime)/duration,1); const ease = 1-Math.pow(1-t,5); rotation = startRotation+(endRotation-startRotation)*ease; drawWheel(); if(t<1) { requestAnimationFrame(animate); } else { rotation = endRotation%(Math.PI*2); drawWheel(); const data = { prize: prizes[winner], time: new Date().toISOString() }; localStorage.setItem(STORAGE_KEY,JSON.stringify(data)); amount.textContent = data.prize; result.style.display = "block"; button.textContent = "🔒 SPIN SUDAH DIGUNAKAN"; spinning = false; } } requestAnimationFrame(animate); });
